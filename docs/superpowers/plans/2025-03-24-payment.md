```python
# -*- coding: utf-8 -*-
"""工行聚合支付客户端"""
import hashlib
import json
import logging
from typing import Optional
from cryptography.hazmat.primitives import hashes, serialization
from cryptography.hazmat.primitives.asymmetric import padding
from cryptography.hazmat.backends import default_backend
from base64 import b64encode, b64decode

logger = logging.getLogger(__name__)


class IcbcClient:
    """工行聚合支付客户端"""

    # API 端点
    API_CREATE_ORDER = "https://gw.open.icbc.com.cn/api/cardbusiness/aggregatepay/b2c/online/consumepurchase/V1"
    API_QUERY_ORDER = "https://gw.open.icbc.com.cn/api/cardbusiness/aggregatepay/b2c/online/orderqry/V1"

    def __init__(
        self,
        app_id: str,
        mer_id: str,
        mer_prtcl_no: str,
        private_key: str,
        public_key: str,
        device_info: str,
        notify_url: str,
    ):
        """
        初始化工行客户端

        Args:
            app_id: 应用编号
            mer_id: 商户编号
            mer_prtcl_no: 协议编号
            private_key: 商户私钥（PEM 格式）
            public_key: 工行公钥（PEM 格式）
            device_info: 设备号
            notify_url: 回调地址
        """
        self.app_id = app_id
        self.mer_id = mer_id
        self.mer_prtcl_no = mer_prtcl_no
        self.private_key = private_key
        self.public_key = public_key
        self.device_info = device_info
        self.notify_url = notify_url

    def _load_private_key(self):
        """加载私钥"""
        from cryptography.hazmat.primitives.asymmetric import rsa
        try:
            # 尝试 PEM 格式
            return serialization.load_pem_private_key(
                self.private_key.encode(),
                password=None,
                backend=default_backend()
            )
        except Exception:
            # 尝试 PKCS8 格式
            return serialization.load_der_private_key(
                b64decode(self.private_key),
                password=None,
                backend=default_backend()
            )

    def _load_public_key(self):
        """加载公钥"""
        try:
            return serialization.load_pem_public_key(
                self.public_key.encode(),
                backend=default_backend()
            )
        except Exception:
            return serialization.load_der_public_key(
                b64decode(self.public_key),
                backend=default_backend()
            )

    def _sign(self, data: str, sign_type: str = "RSA2") -> str:
        """
        RSA2 签名

        Args:
            data: 待签名字符串
            sign_type: 签名类型（RSA2/RSA）

        Returns:
            Base64 编码的签名
        """
        private_key = self._load_private_key()

        if sign_type == "RSA2":
            # RSA256 签名
            signature = private_key.sign(
                data.encode('utf-8'),
                padding.PKCS1v15(),
                hashes.SHA256()
            )
        else:
            # RSA 签名（SHA1）
            signature = private_key.sign(
                data.encode('utf-8'),
                padding.PKCS1v15(),
                hashes.SHA1()
            )

        return b64encode(signature).decode('utf-8')

    def _verify(self, data: str, sign: str, sign_type: str = "RSA") -> bool:
        """
        RSA 验签

        Args:
            data: 待验签字符串
            sign: Base64 编码的签名
            sign_type: 签名类型（RSA2/RSA）

        Returns:
            验签结果
        """
        public_key = self._load_public_key()
        signature = b64decode(sign)

        try:
            if sign_type == "RSA2":
                public_key.verify(
                    signature,
                    data.encode('utf-8'),
                    padding.PKCS1v15(),
                    hashes.SHA256()
                )
            else:
                public_key.verify(
                    signature,
                    data.encode('utf-8'),
                    padding.PKCS1v15(),
                    hashes.SHA1()
                )
            return True
        except Exception as e:
            logger.error(f"验签失败: {e}")
            return False

    def _build_sign_str(self, params: dict, path: str = "") -> str:
        """
        构建签名字符串

        工行签名规则：参数按字典序排序，拼接成 key1=value1&key2=value2 格式
        """
        # 过滤空值和 sign 字段
        filtered = {k: v for k, v in params.items() if v is not None and v != "" and k != "sign"}
        # 按字典序排序
        sorted_params = sorted(filtered.items())
        # 拼接
        sign_str = "&".join([f"{k}={v}" for k, v in sorted_params])
        return sign_str
```

```python
import httpx

async def create_order(
    self,
    out_trade_no: str,
    amount: int,
    expire_time: int = 900,
    body: str = "积分充值",
) -> dict:
    """
    调用工行统一下单接口

    Args:
        out_trade_no: 商户订单号
        amount: 金额（分）
        expire_time: 超时时间（秒），默认900秒
        body: 商品描述

    Returns:
        工行响应，包含 pay_url 等字段
    """
    from datetime import datetime

    # 构建业务参数
    biz_content = {
        "mer_id": self.mer_id,
        "mer_prtcl_no": self.mer_prtcl_no,
        "out_trade_no": out_trade_no,
        "orig_date_time": datetime.now().strftime("%Y-%m-%dT%H:%M:%S"),
        "fee_type": "001",  # 人民币
        "total_fee": str(amount),
        "pay_mode": "9",  # 微信
        "access_type": "9",  # 微信小程序
        "notify_type": "HS",  # 主动通知
        "result_type": "0",  # 无论成功失败都通知
        "expire_time": str(expire_time),
        "body": body,
        "mer_url": self.notify_url,
        "device_info": self.device_info,
        "icbc_appid": self.app_id,
    }

    # 构建请求参数
    from datetime import datetime
    timestamp = datetime.now().strftime("%Y-%m-%d %H:%M:%S")

    params = {
        "app_id": self.app_id,
        "format": "json",
        "charset": "UTF-8",
        "sign_type": "RSA2",
        "timestamp": timestamp,
        "biz_content": json.dumps(biz_content, ensure_ascii=False),
    }

    # 签名
    sign_str = self._build_sign_str(params)
    sign = self._sign(sign_str, "RSA2")
    params["sign"] = sign

    # 发送请求
    async with httpx.AsyncClient(timeout=30) as client:
        response = await client.post(
            self.API_CREATE_ORDER,
            json=params,
            headers={"Content-Type": "application/json"}
        )
        response.raise_for_status()
        result = response.json()

    logger.info(f"工行下单响应: {result}")
    return result
```

```python
# -*- coding: utf-8 -*-
"""支付服务"""
import logging
import uuid
from datetime import datetime, timedelta
from typing import Optional
from sqlalchemy.orm import Session
from ..db_models import (
    PaymentOrderModel, PointTransactionModel, PaymentOrderStatus,
    PointTransactionType, PointSourceType, EnterpriseModel
)
from .icbc_client import IcbcClient
from .point_service import PointService

logger = logging.getLogger(__name__)


class PaymentService:
    """支付服务"""

    # 订单超时时间（秒）
    ORDER_EXPIRE_TIME = 900  # 15分钟

    # 金额限制（分）
    MIN_AMOUNT = 100  # 1元
    MAX_AMOUNT = 500000  # 5000元

    def __init__(self, db: Session, icbc_client: IcbcClient, point_service: PointService):
        self.db = db
        self.icbc_client = icbc_client
        self.point_service = point_service

    def _generate_out_trade_no(self) -> str:
        """生成商户订单号"""
        import time
        timestamp = time.strftime("%Y%m%d%H%M%S")
        random_str = uuid.uuid4().hex[:6].upper()
        return f"PO{timestamp}{random_str}"

    def _get_user_enterprise(self, user_id: str) -> Optional[EnterpriseModel]:
        """获取用户所属企业"""
        from ..db_models import UserModel
        user = self.db.query(UserModel).filter(
            UserModel.user_id == user_id
        ).first()
        if not user or not user.enterprise_id:
            return None
        return self.db.query(EnterpriseModel).filter(
            EnterpriseModel.id == user.enterprise_id
        ).first()

    async def create_payment_order(
        self,
        user_id: str,
        amount: int
    ) -> PaymentOrderModel:
        """
        创建支付订单

        Args:
            user_id: 用户ID
            amount: 金额（分）

        Returns:
            支付订单
        """
        # 金额校验
        if amount < self.MIN_AMOUNT or amount > self.MAX_AMOUNT:
            raise ValueError(f"金额必须在{self.MIN_AMOUNT}-{self.MAX_AMOUNT}分之间")

        # 获取用户企业
        enterprise = self._get_user_enterprise(user_id)
        if not enterprise:
            raise ValueError("用户未关联企业")

        # 创建支付订单
        out_trade_no = self._generate_out_trade_no()
        expire_at = datetime.now() + timedelta(seconds=self.ORDER_EXPIRE_TIME)

        order = PaymentOrderModel(
            id=str(uuid.uuid4()),
            user_id=user_id,
            out_trade_no=out_trade_no,
            amount=amount,
            status=PaymentOrderStatus.created,
            expire_at=expire_at,
        )

        self.db.add(order)
        self.db.flush()

        try:
            # 调用工行下单
            icbc_response = await self.icbc_client.create_order(
                out_trade_no=out_trade_no,
                amount=amount,
                expire_time=self.ORDER_EXPIRE_TIME,
            )

            # 更新订单状态
            order.status = PaymentOrderStatus.processing
            order.icbc_response = icbc_response
            order.submitted_at = datetime.now()

            # 解析支付URL
            if icbc_response.get("return_code") == 0:
                biz_content = icbc_response.get("biz_content", {})
                order.pay_url = biz_content.get("pay_url")
                order.qr_code_data = biz_content.get("pay_url")  # 二维码数据与 pay_url 相同

            self.db.commit()
            self.db.refresh(order)

            logger.info(f"创建支付订单成功: {out_trade_no}")
            return order

        except Exception as e:
            self.db.rollback()
            order.status = PaymentOrderStatus.failed
            self.db.commit()
            logger.error(f"创建支付订单失败: {e}")
            raise

    async def query_payment_status(self, order_id: str) -> dict:
        """
        查询支付状态

        Args:
            order_id: 订单ID

        Returns:
            订单状态信息
        """
        order = self.db.query(PaymentOrderModel).filter(
            PaymentOrderModel.id == order_id
        ).first()

        if not order:
            raise ValueError("订单不存在")

        # 如果订单是 processing 状态，主动查询工行
        if order.status == PaymentOrderStatus.processing:
            try:
                icbc_response = await self.icbc_client.query_order(order.out_trade_no)
                biz_content = icbc_response.get("biz_content", {})
                order_status = biz_content.get("order_status")

                if order_status == "1":  # 已支付
                    await self._handle_payment_success(order, icbc_response)
                elif order_status == "2":  # 已取消
                    order.status = PaymentOrderStatus.cancelled
                    self.db.commit()

            except Exception as e:
                logger.error(f"查询工行订单失败: {e}")

        return {
            "id": order.id,
            "out_trade_no": order.out_trade_no,
            "status": order.status.value,
            "amount": order.amount,
            "paid_at": order.paid_at.isoformat() if order.paid_at else None,
            "expire_at": order.expire_at.isoformat() if order.expire_at else None,
        }

    async def handle_notify(self, notify_data: dict) -> dict:
        """
        处理工行回调

        Args:
            notify_data: 回调数据

        Returns:
            应答工行的响应
        """
        biz_content = notify_data.get("biz_content")
        if isinstance(biz_content, str):
            biz_content = json.loads(biz_content)

        out_trade_no = biz_content.get("out_trade_no")
        return_code = biz_content.get("return_code")

        # 查找订单
        order = self.db.query(PaymentOrderModel).filter(
            PaymentOrderModel.out_trade_no == out_trade_no
        ).first()

        if not order:
            logger.error(f"回调订单不存在: {out_trade_no}")
            return {"return_code": -1, "return_msg": "订单不存在"}

        # 记录回调信息
        order.notified_at = datetime.now()
        order.notify_data = notify_data

        # 检查是否已处理
        if order.status == PaymentOrderStatus.paid:
            return {"return_code": 0, "return_msg": "success"}

        # 处理支付结果
        if return_code == "0":  # 支付成功
            await self._handle_payment_success(order, biz_content)
            return {"return_code": 0, "return_msg": "success"}
        else:
            order.status = PaymentOrderStatus.failed
            self.db.commit()
            return {"return_code": -1, "return_msg": "支付失败"}

    async def _handle_payment_success(self, order: PaymentOrderModel, icbc_response: dict):
        """处理支付成功"""
        biz_content = icbc_response if isinstance(icbc_response, dict) else {}
        if not biz_content and "biz_content" in str(icbc_response):
            biz_content = json.loads(icbc_response.get("biz_content", "{}"))

        order.status = PaymentOrderStatus.paid
        order.paid_at = datetime.now()
        order.third_trade_no = biz_content.get("third_trade_no")

        # 获取积分兑换比例
        points_per_yuan = self._get_points_per_yuan()
        points = order.amount * points_per_yuan // 100

        # 添加积分
        enterprise = self._get_user_enterprise(order.user_id)
        if enterprise:
            transaction = await self.point_service.add_points(
                enterprise_id=enterprise.id,
                amount=points,
                source_type=PointSourceType.online_payment,
                remark=f"支付充值，订单号: {order.out_trade_no}"
            )

            # 关联支付订单
            transaction.payment_id = order.id
            transaction.amount = order.amount
            self.db.commit()

        logger.info(f"支付成功: {order.out_trade_no}, 积分: {points}")

    def _get_points_per_yuan(self) -> int:
        """获取积分兑换比例"""
        from ..db_models import SystemConfigModel
        config = self.db.query(SystemConfigModel).filter(
            SystemConfigModel.key == "points_per_yuan"
        ).first()
        return int(config.value) if config else 100

    def mark_order_timeout(self, order_id: str):
        """标记订单超时"""
        order = self.db.query(PaymentOrderModel).filter(
            PaymentOrderModel.id == order_id
        ).first()

        if order and order.status == PaymentOrderStatus.processing:
            order.status = PaymentOrderStatus.timeout
            self.db.commit()
            logger.info(f"订单超时: {order.out_trade_no}")

    def get_recharge_records(
        self,
        user_id: str,
        page: int = 1,
        page_size: int = 20
    ) -> tuple:
        """
        获取充值记录

        Args:
            user_id: 用户ID
            page: 页码
            page_size: 每页数量

        Returns:
            (记录列表, 总数)
        """
        # 获取用户企业
        enterprise = self._get_user_enterprise(user_id)
        if not enterprise:
            return [], 0

        query = self.db.query(PointTransactionModel).filter(
            PointTransactionModel.enterprise_id == enterprise.id,
            PointTransactionModel.payment_id.isnot(None),  # 只看充值记录
        )

        total = query.count()

        records = query.order_by(
            PointTransactionModel.created_at.desc()
        ).offset((page - 1) * page_size).limit(page_size).all()

        return records, total
```

```python
# -*- coding: utf-8 -*-
"""支付定时任务"""
import logging
from apscheduler.schedulers.asyncio import AsyncIOScheduler
from sqlalchemy.orm import Session
from sqlalchemy import create_engine
from ..database import DATABASE_URL, SessionLocal
from ..db_models import PaymentOrderModel, PaymentOrderStatus
from datetime import datetime

logger = logging.getLogger(__name__)

scheduler = AsyncIOScheduler()


async def check_expired_orders():
    """检查超时订单"""
    db = SessionLocal()

    try:
        # 查找超时的 processing 订单
        expired_orders = db.query(PaymentOrderModel).filter(
            PaymentOrderModel.status == PaymentOrderStatus.processing,
            PaymentOrderModel.expire_at < datetime.now()
        ).all()

        count = 0
        for order in expired_orders:
            order.status = PaymentOrderStatus.timeout
            count += 1

        db.commit()

        if count > 0:
            logger.info(f"标记 {count} 个订单为超时")

    except Exception as e:
        logger.error(f"检查超时订单失败: {e}")
        db.rollback()
    finally:
        db.close()


def start_scheduler():
    """启动定时任务"""
    # 每分钟检查一次
    scheduler.add_job(
        check_expired_orders,
        'interval',
        minutes=1,
        id='check_expired_orders',
        replace_existing=True
    )
    scheduler.start()
    logger.info("支付定时任务已启动")
```
